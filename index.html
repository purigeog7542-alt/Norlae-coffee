create table coffee_sales ( id bigint generated by default as identity primary key, created_at timestamp with time zone default timezone('utc'::text, now()) not null, customer_type text not null, product_name text not null, quantity int not null, total_price numeric not null ); -- ตั้งค่าให้ใช้งานแบบ Public เพื่อให้ระบบ POS บันทึกข้อมูลได้ทันที alter table coffee_sales enable row level security; create policy "Enable insert for all users" on coffee_sales for insert with check (true); create policy "Enable select for all users" on coffee_sales for select using (true);
